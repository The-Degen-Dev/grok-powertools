# Live Browser UI QA Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Run a full live browser QA pass across the local web app, extension popup, live Grok overlay, and real Grok image/video generation without touching backup or R2 write paths.

**Architecture:** Treat this as a controlled QA run, not a product-code change. Use existing repo scripts for gates, a temporary Playwright script for local web UI checks, and the user's existing Chrome session plus Peekaboo/AppleScript for authenticated Grok. Store redacted screenshots, JSON metrics, and a final report under one run-scoped directory in `/private/tmp`.

**Tech Stack:** Node 24 through `mise`, npm, Playwright, Next.js local dev server, AppleScript for existing Chrome tab targeting, Peekaboo for native Chrome screenshots and keyboard input, Chrome MV3 extension loaded in the user's existing Chrome profile.

---

## Scope Check

This is one QA execution plan, not a backup acceptance plan. It crosses local web, Chrome extension, and live Grok UI because those surfaces are the product experience being tested together.

This plan intentionally excludes:

- Full Media Backup.
- One Media R2 Canary.
- Test Upload Pipeline.
- Run Backfill.
- Retry Unsynced.
- Start Sync.
- Production or acceptance R2 writes.
- Destructive reset of processed IDs, saved prompts, prompt history, or Chrome storage.

If an execution worker needs to test any backup path, stop and write a separate plan from `docs/superpowers/specs/2026-06-09-live-acceptance-e2e-design.md`.

  checkedAt: new Date().toISOString(),

  flows: [],
  console: [],
  pageErrors: [],
  failedRequests: [],
  badResponses: [],
  screenshots: []

async function newPage(viewport, label) {
  const context = await browser.newContext({ viewport });
  const page = await context.newPage();
  page.on("console", (msg) => evidence.console.push({ label, type: msg.type(), text: msg.text().slice(0, 500), url: page.url() }));
  page.on("pageerror", (err) => evidence.pageErrors.push({ label, message: String(err.message || err).slice(0, 500), url: page.url() }));
  page.on("requestfailed", (req) => evidence.failedRequests.push({ label, url: req.url(), failure: req.failure()?.errorText || "unknown" }));
  page.on("response", (res) => {
    if (res.status() >= 500) evidence.badResponses.push({ label, url: res.url(), status: res.status() });
  });
  return { context, page };
}

async function shot(page, name) {
  const path = `${dir}/web-ui-${name}.png`;
  await page.screenshot({ path, fullPage: true });
  evidence.screenshots.push(path);
}

async function metrics(page, label) {
  const m = await page.evaluate(() => {
    const rects = [...document.querySelectorAll("body *")]
      .map((el) => {
        const r = el.getBoundingClientRect();
        const text = (el.textContent || "").trim().replace(/\s+/g, " ").slice(0, 80);
        return { tag: el.tagName, text, x: r.x, y: r.y, w: r.width, h: r.height };
      })
      .filter((r) => r.w > 0 && r.h > 0 && (r.x + r.w > window.innerWidth + 2 || r.x < -2 || r.y < -2));
    return {
      url: location.href,
      title: document.title,
      viewport: { width: window.innerWidth, height: window.innerHeight },
      scrollWidth: document.documentElement.scrollWidth,
      clientWidth: document.documentElement.clientWidth,
      horizontalOverflow: document.documentElement.scrollWidth > document.documentElement.clientWidth + 2,
      offscreen: rects.slice(0, 20),
      visibleText: document.body.innerText.slice(0, 1000)
    };
  });
  evidence.flows.push({ label, ...m });
  return m;
}

async function assertModalInViewport(page, label, text) {
  const modal = page.getByText(text, { exact: false }).first();
  await modal.waitFor({ state: "visible", timeout: 5000 });
  const box = await modal.evaluate((el) => {
    const root = el.closest("[role='dialog']") || el.closest("aside") || el.parentElement;
    const r = (root || el).getBoundingClientRect();
    return { x: r.x, y: r.y, w: r.width, h: r.height, vw: window.innerWidth, vh: window.innerHeight };
  });
  evidence.flows.push({ label, modal: box });
  if (box.x < -2 || box.y < -2 || box.x + box.w > box.vw + 2 || box.y + box.h > box.vh + 2) {
    throw new Error(`${label} modal outside viewport: ${JSON.stringify(box)}`);
  }
}

await desktop.page.goto(base, { waitUntil: "networkidle" });

await metrics(desktop.page, "desktop-home-initial");

await desktop.page.goto(`${base}/movie`, { waitUntil: "networkidle" });
await shot(desktop.page, "desktop-movie");
await metrics(desktop.page, "desktop-movie");

await desktop.page.goto(`${base}/edit?video=${encodeURIComponent(videoUrl)}`, { waitUntil: "networkidle" });
await desktop.page.waitForFunction(() => {
  const v = document.querySelector("video");
  return v && v.readyState >= 2 && Number.isFinite(v.duration) && v.duration > 0;
}, { timeout: 20000 });
const editorVideo = await desktop.page.evaluate(() => {
  const v = document.querySelector("video");
  return { readyState: v?.readyState, duration: v?.duration, src: v?.currentSrc, text: document.body.innerText.slice(0, 800) };
});
evidence.flows.push({ label: "desktop-editor-valid-video", video: editorVideo });
await shot(desktop.page, "desktop-editor-valid-video");

await desktop.page.goto(`${base}/edit?video=${encodeURIComponent("https://example.invalid/video.mp4")}`, { waitUntil: "networkidle" });
